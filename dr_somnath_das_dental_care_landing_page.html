Supabase Database & SQL Setup

To store appointments and patient reviews permanently in Supabase when hosted on GitHub Pages:

Step 1: Connect Your Supabase Project

You can paste your credentials below to test immediately, or hardcode them into SUPABASE_URL & SUPABASE_ANON_KEY inside index.html.

Step 2: Run This SQL in Supabase SQL Editor 
-- 1. Create Appointments Table
CREATE TABLE IF NOT EXISTS public.appointments (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    patient_name TEXT NOT NULL,
    phone_number TEXT NOT NULL,
    preferred_date DATE NOT NULL,
    preferred_session TEXT NOT NULL,
    treatment TEXT NOT NULL,
    notes TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. Create Patient Reviews Table
CREATE TABLE IF NOT EXISTS public.reviews (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    reviewer_name TEXT NOT NULL,
    rating INTEGER NOT NULL CHECK (rating >= 1 AND rating <= 5),
    treatment TEXT,
    comment TEXT NOT NULL,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 3. Enable Row Level Security (RLS) for Static GitHub Pages
ALTER TABLE public.appointments ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.reviews ENABLE ROW LEVEL SECURITY;

-- Allow anyone to book an appointment
CREATE POLICY "Allow public appointment insert" ON public.appointments
    FOR INSERT TO anon WITH CHECK (true);

-- Allow anyone to read & submit reviews
CREATE POLICY "Allow public reviews select" ON public.reviews
    FOR SELECT TO anon USING (true);

CREATE POLICY "Allow public reviews insert" ON public.reviews
    FOR INSERT TO anon WITH CHECK (true);
